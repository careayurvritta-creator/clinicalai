// Add retry logic with exponential backoff
async function createChatStreamWithRetry(messages, model, retries = 3) {
  for (let i = 0; i < retries; i++) {
    try {
      return await createChatStream(messages, model)
    } catch (error) {
      if (i === retries - 1) throw error
      await sleep(Math.pow(2, i) * 1000)
    }
  }
}

// Connection pooling
const clientPool = new Map<string, NVIDIA>()
function getClient(apiKey: string): NVIDIA {
  if (!clientPool.has(apiKey)) {
    clientPool.set(apiKey, new NVIDIA({ apiKey }))
  }
  return clientPool.get(apiKey)!
